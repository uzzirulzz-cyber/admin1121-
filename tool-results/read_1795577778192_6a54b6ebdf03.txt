   672→function CheckoutDialog({
   673→  open,
   674→  items,
   675→  totalCents,
   676→  onClose,
   677→  onSuccess,
   678→}: {
   679→  open: boolean;
   680→  items: CartItem[];
   681→  totalCents: number;
   682→  onClose: () => void;
   683→  onSuccess: (result: CheckoutResult) => void;
   684→}) {
   685→  const { toast } = useToast();
   686→  const [name, setName] = useState("");
   687→  const [email, setEmail] = useState("");
   688→  const [notes, setNotes] = useState("");
   689→  const [submitting, setSubmitting] = useState(false);
   690→
   691→  async function submit(e: React.FormEvent) {
   692→    e.preventDefault();
   693→    if (items.length === 0) return;
   694→    setSubmitting(true);
   695→    try {
   696→      const result = await api<CheckoutResult>("/api/public/orders", {
   697→        method: "POST",
   698→        body: {
   699→          customerName: name,
   700→          customerEmail: email,
   701→          notes: notes || null,
   702→          items: items.map((i) => ({ productId: i.productId, licenseType: i.licenseType })),
   703→        },
   704→      });
   705→      onSuccess(result);
   706→      setName("");
   707→      setEmail("");
   708→      setNotes("");
   709→    } catch (err) {
   710→      toast({
   711→        title: "Checkout failed",
   712→        description: err instanceof Error ? err.message : "Please try again",
   713→      });
   714→    } finally {
   715→      setSubmitting(false);
   716→    }
   717→  }
   718→
   719→  return (
   720→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   721→      <DialogContent className="bg-card max-w-lg max-h-[85vh] overflow-y-auto">
   722→        <DialogHeader>
   723→          <DialogTitle>Checkout</DialogTitle>
   724→          <DialogDescription>
   725→            You pay after the order: we send instructions and an admin confirms your payment.
   726→          </DialogDescription>
   727→        </DialogHeader>
   728→        <form onSubmit={submit} className="flex flex-col gap-4 mt-2">
   729→          <div className="grid gap-2">
   730→            <Label htmlFor="co-name">Full name</Label>
   731→            <Input
   732→              id="co-name"
   733→              value={name}
   734→              onChange={(e) => setName(e.target.value)}
   735→              placeholder="Your name"
   736→              required
   737→              minLength={1}
   738→            />
   739→          </div>
   740→          <div className="grid gap-2">
   741→            <Label htmlFor="co-email">Email</Label>
   742→            <Input
   743→              id="co-email"
   744→              type="email"
   745→              value={email}
   746→              onChange={(e) => setEmail(e.target.value)}
   747→              placeholder="you@example.com"
   748→              required
   749→            />
   750→          </div>
   751→          <div className="grid gap-2">
   752→            <Label htmlFor="co-notes">Notes (optional)</Label>
   753→            <Textarea
   754→              id="co-notes"
   755→              value={notes}
   756→              onChange={(e) => setNotes(e.target.value)}
   757→              placeholder="Anything we should know? Social links, deadline, etc."
   758→              rows={2}
   759→            />
   760→          </div>
   761→          <div className="rounded-lg border border-border p-3 text-sm flex flex-col gap-1">
   762→            {items.map((i) => (
   763→              <div key={`${i.productId}-${i.licenseType}`} className="flex justify-between">
   764→                <span className="truncate mr-2">
   765→                  {i.title} <span className="text-muted-foreground">({i.licenseType})</span>
   766→                </span>
   767→                <span className="font-medium">{formatMoney(i.priceCents)}</span>
   768→              </div>
   769→            ))}
   770→            <Separator className="my-1" />
   771→            <div className="flex justify-between font-semibold">
   772→              <span>Total</span>
   773→              <span>{formatMoney(totalCents)}</span>
   774→            </div>
   775→          </div>
   776→          <Button type="submit" className="h-11" disabled={submitting || items.length === 0}>
   777→            {submitting ? "Placing order..." : "Place order"}
   778→          </Button>
   779→        </form>
   780→      </DialogContent>
   781→    </Dialog>
   782→  );
   783→}

   797→function PlayerInner({ track }: { track: { trackId: string; title: string; coverUrl: string | null; audioUrl: string | null } }) {
   798→  const audioRef = useRef<HTMLAudioElement | null>(null);
   799→  const playing = usePlayer((s) => s.playing);
   800→  const toggle = usePlayer((s) => s.toggle);
   801→  const stop = usePlayer((s) => s.stop);
   802→  const setPlaying = usePlayer((s) => s.setPlaying);
   803→  const [progress, setProgress] = useState(0);
   804→
   805→  useEffect(() => {
   806→    const audio = audioRef.current;
   807→    if (!audio) return;
   808→    if (playing) {
   809→      audio.play().catch(() => setPlaying(false));
   810→    } else {
   811→      audio.pause();
   812→    }
   813→  }, [playing, setPlaying]);
   814→
   815→  return (
   816→    <>
   817→      <audio
   818→        ref={audioRef}
   819→        src={track.audioUrl ?? undefined}
   820→        onEnded={() => setPlaying(false)}
   821→        onTimeUpdate={(e) => {
   822→          const audio = e.currentTarget;
   823→          if (audio.duration) setProgress(audio.currentTime / audio.duration);
   824→        }}
   825→        hidden
   826→      />
   827→      <div className="max-w-6xl mx-auto px-4 sm:px-6 py-3 flex items-center gap-4">
   828→        {track.coverUrl ? (
   829→          <img
   830→            src={track.coverUrl}
   831→            alt={`${track.title} cover`}
   832→            className="h-11 w-11 rounded-md object-cover"
   833→          />
   834→        ) : (
   835→          <div className="h-11 w-11 rounded-md bg-muted flex items-center justify-center">
   836→            <Music2 className="h-4 w-4 text-muted-foreground" />
   837→          </div>
   838→        )}
   839→        <div className="min-w-0 w-40 sm:w-56">
   840→          <p className="text-sm font-medium truncate">{track.title}</p>
   841→          <p className="text-xs text-muted-foreground">Preview</p>
   842→        </div>
   843→        <Button
   844→          size="icon"
   845→          className="rounded-full h-10 w-10 shrink-0"
   846→          onClick={toggle}
   847→          aria-label={playing ? "Pause preview" : "Play preview"}
   848→        >
   849→          {playing ? <Pause className="h-5 w-5" /> : <Play className="h-5 w-5 ml-0.5" />}
   850→        </Button>
   851→        <Slider
   852→          value={[progress * 100]}
   853→          max={100}
   854→          step={0.5}
   855→          onValueChange={(v) => {
   856→            const audio = audioRef.current;
   857→            if (audio && audio.duration) {
   858→              audio.currentTime = (v[0] / 100) * audio.duration;
   859→            }
   860→            setProgress(v[0] / 100);
   861→          }}
   862→          className="flex-1"
   863→          aria-label="Seek position"
   864→        />
   865→        <Button
   866→          variant="ghost"
   867→          size="icon"
   868→          onClick={stop}
   869→          aria-label="Close player"
   870→          className="shrink-0"
   871→        >
   872→          <X className="h-4 w-4" />
   873→        </Button>
   874→      </div>
   875→    </>
   876→  );
   877→}
